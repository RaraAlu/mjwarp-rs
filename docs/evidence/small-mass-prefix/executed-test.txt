#[test]
#[cfg(feature = "cuda-probe")]
#[ignore = "opt-in CUDA-feature hash certificate; no device or author execution"]
fn gpu_frozen_reference_hash_certificate() {
    let manifest: Value =
        serde_json::from_str(include_str!("../fixtures/small-mass-prefix/manifest.json")).unwrap();
    let contract: Value =
        serde_json::from_str(include_str!("../fixtures/small-mass-prefix/contract.json")).unwrap();
    let hash = |p: &str| {
        format!(
            "{:x}",
            Sha256::digest(
                std::fs::read(std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join(p)).unwrap()
            )
        )
    };
    assert_eq!(
        hash("fixtures/small-mass-prefix/reference.json"),
        manifest["output_sha256"]
    );
    assert_eq!(
        hash("fixtures/small-mass-prefix/contract.json"),
        manifest["contract_sha256"]
    );
    for pin in contract["files"].as_array().unwrap() {
        assert_eq!(hash(pin["path"].as_str().unwrap()), pin["sha256"]);
    }
}
#[test]
fn freezes_independent_known_u_and_pairs_existing_native_fractional_cases() {
    let refs = input::analytic();
    let model = input::mixed_input();
    assert_eq!(
        (model.nv(), model.nc(), model.qld_block_total()),
        (86, 2201, 90)
    );
    for sample in 0..2 {
        assert_eq!(input::mixed_matrix(sample).len(), 2201);
        let mut csr = 0;
        for block in refs["blocks"].as_array().unwrap() {
            let n = block["size"].as_u64().unwrap() as usize;
            let a = block["address"].as_u64().unwrap() as usize;
            let upper = &refs["cases"][sample]["upper"];
            for i in 0..n {
                for j in 0..=i {
                    let gram = (0..n)
                        .map(|k| {
                            upper[a + k * n + i].as_f64().unwrap()
                                * upper[a + k * n + j].as_f64().unwrap()
                        })
                        .sum::<f64>();
                    assert_eq!(gram, refs["cases"][sample]["csr"][csr].as_f64().unwrap());
                    assert_eq!(gram, f64::from(gram as f32));
                    csr += 1;
                }
            }
        }
        assert_eq!(csr, 55);
    }
    let (native, frozen) = native_refs();
    let m = frozen["models"]
        .as_array()
        .unwrap()
        .iter()
        .find(|m| m["name"] == "armature-chain")
        .unwrap();
    for sample in 0..8 {
        let row = m["rows"]
            .as_array()
            .unwrap()
            .iter()
            .find(|r| r["case"] == sample)
            .unwrap();
        for j in 0..3 {
            assert_eq!(
                native["cases"][sample]["qpos"][j].as_f64(),
                row["qpos"][j].as_f64()
            );
        }
    }
    assert_eq!(native_input().qld_block_total(), 4);
}

    fn compare(
        out: &SparseMassSmallOutput,
        w: usize,
        matrix: &[f32],
        expected: &[f64],
    ) -> (f64, f64) {
        let v = out.world(w).unwrap();
        assert_eq!(v.len(), expected.len());
        let mut max = 0.0f64;
        let mut reconstruction = 0.0f64;
        for (&a, &b) in v.iter().zip(expected) {
            let error = (f64::from(a) - b).abs();
            assert!(error <= 2e-4 + 2e-4 * b.abs());
            max = max.max(error);
        }
        let f = out.input().fields();
        for block in out
            .input()
            .blocks()
            .iter()
            .filter(|b| b.route == SparseMassRoute::SmallCholesky)
        {
            let a = out.input().qld_block_adr()[block.start] as usize;
            let n = block.size;
            let mut error = 0.0f64;
            let mut norm = 0.0f64;
            for i in 0..n {
                let mut row_error = 0.0;
                let mut row_norm = 0.0;
                for j in 0..n {
                    if i > j {
                        assert_eq!(v[a + i * n + j].to_bits(), 0.0f32.to_bits());
                    }
                    let row = block.start + i.max(j);
                    let col = block.start + i.min(j);
                    let k = (f.m_rowadr[row] as usize
                        ..(f.m_rowadr[row] + f.m_rownnz[row]) as usize)
                        .find(|&k| f.m_colind[k] as usize == col)
                        .unwrap();
                    let m = f64::from(matrix[k]);
                    let actual = (0..n)
                        .map(|k| f64::from(v[a + k * n + i]) * f64::from(v[a + k * n + j]))
                        .sum::<f64>();
                    row_error += (actual - m).abs();
                    row_norm += m.abs();
                }
                error = error.max(row_error);
                norm = norm.max(row_norm);
            }
            assert!(error / norm <= 2e-6, "reconstruction={}", error / norm);
            reconstruction = reconstruction.max(error / norm);
        }
        (max, reconstruction)
    }
    fn bits(v: &[f32]) -> Vec<u32> {
        v.iter().map(|v| v.to_bits()).collect()
    }
    #[test]
    #[ignore = "requires NVIDIA/NVRTC; new constant-size Small prefix factor purpose"]
    fn gpu_analytic_and_native_small_prefixes_follow_fixed_tolerances() {
        let (native, frozen) = native_refs();
        let native_expected = frozen["models"]
            .as_array()
            .unwrap()
            .iter()
            .find(|m| m["name"] == "armature-chain")
            .unwrap();
        let analytic = input::analytic();
        let session = TransferSession::new(0).unwrap();
        let mut batches = Vec::new();
        let mut total = 0;
        let mut max = 0.0f64;
        let mut reconstruction = 0.0f64;
        for mode in 0..2 {
            let model = if mode == 0 {
                input::mixed_input()
            } else {
                native_input()
            };
            let prefix = model.qld_block_total();
            let nc = model.nc();
            let states = if mode == 0 { 2 } else { 8 };
            let plan = SparseMassSmallPlan::new(&session, model).unwrap();
            for worlds in [1, 2, 257] {
                let start = std::time::Instant::now();
                let mut data = plan.create_data(worlds).unwrap();
                for sample in 0..states {
                    let (matrix, expected): (Vec<f32>, Vec<f64>) = if mode == 0 {
                        (
                            input::mixed_matrix(sample),
                            analytic["cases"][sample]["upper"]
                                .as_array()
                                .unwrap()
                                .iter()
                                .map(|v| v.as_f64().unwrap())
                                .collect(),
                        )
                    } else {
                        let raw = &native["cases"][sample]["matrix"];
                        let row = native_expected["rows"]
                            .as_array()
                            .unwrap()
                            .iter()
                            .find(|r| r["case"] == sample)
                            .unwrap();
                        (
                            [0, 3, 4, 8]
                                .map(|k| raw[k].as_f64().unwrap() as f32)
                                .to_vec(),
                            [0, 1, 3, 4]
                                .map(|k| row["native_factor"][k].as_f64().unwrap())
                                .to_vec(),
                        )
                    };
                    data.write_matrix(&matrix.repeat(worlds)).unwrap();
                    plan.update(&mut data).unwrap();
                    let out = data.readback().unwrap();
                    for w in 0..worlds {
                        let (a, r) = compare(&out, w, &matrix, &expected);
                        max = max.max(a);
                        reconstruction = reconstruction.max(r);
                        total += prefix;
                    }
                }
                let descriptor_words = if mode == 0 { 15 } else { 3 };
                batches.push(json!({"mode":mode,"worlds":worlds,"states":states,"prefix":prefix,"slots":worlds*states*prefix,"phase_elapsed_ms":start.elapsed().as_millis(),"device_buffer_bytes":4*((nc+2*prefix+1)*worlds+32+prefix.max(1)+descriptor_words)}));
            }
        }
        let result = json!({"schema":1,"purpose":"new Small2..6 U prefix only; no Tile/Sparse/Compact factors","factor_slots":total,"max_absolute":max,"max_normalized_reconstruction":reconstruction,"batches":batches,"actual_frozen_sizes":"Small2 source/PTX checked; Small3..6 actual generated comparison unaccepted","timing_includes_host_compare":true});
        if let Some(dir) = std::env::var_os("MJWARP_SMALL_PREFIX_RESULTS") {
            std::fs::create_dir_all(&dir).unwrap();
            std::fs::write(
                std::path::Path::new(&dir).join("results.json"),
                serde_json::to_vec_pretty(&result).unwrap(),
            )
            .unwrap();
        }
        println!("{result}");
    }
    #[test]
    #[ignore = "requires NVIDIA/NVRTC; new Small prefix cache/world/pivot rejection purpose"]
    fn gpu_cache_world_writes_and_pivot_recovery_preserve_existing_admission() {
        let session = TransferSession::new(0).unwrap();
        let plan = SparseMassSmallPlan::new(&session, input::mixed_input()).unwrap();
        assert!(plan.create_data(0).is_err());
        let mut data = plan.create_data(257).unwrap();
        let m0 = input::mixed_matrix(0);
        let m1 = input::mixed_matrix(1);
        data.write_matrix(&m0.repeat(257)).unwrap();
        assert!(plan.update_factor(&mut data).is_err());
        plan.update(&mut data).unwrap();
        let before = data.readback().unwrap();
        let foreign = SparseMassSmallPlan::new(&session, input::mixed_input()).unwrap();
        assert!(foreign.update_factor(&mut data).is_err());
        assert!(data.write_matrix(&[]).is_err());
        assert!(data.write_world_matrix(257, &m0).is_err());
        assert!(
            data.write_world_matrix(0, &vec![f32::NAN; m0.len()])
                .is_err()
        );
        assert_eq!(
            bits(data.readback().unwrap().world(0).unwrap()),
            bits(before.world(0).unwrap())
        );
        data.write_world_matrix(256, &m1).unwrap();
        assert!(data.readback().is_err());
        assert!(plan.update_factor(&mut data).is_err());
        plan.update_gather(&mut data).unwrap();
        plan.update_factor(&mut data).unwrap();
        let after = data.readback().unwrap();
        for w in [0, 255] {
            assert_eq!(
                bits(after.world(w).unwrap()),
                bits(before.world(w).unwrap())
            );
        }
        let expected: Vec<f64> = input::analytic()["cases"][1]["upper"]
            .as_array()
            .unwrap()
            .iter()
            .map(|v| v.as_f64().unwrap())
            .collect();
        compare(&after, 256, &m1, &expected);
        let mut bad = m1.clone();
        bad[2145] = -1.0;
        data.write_world_matrix(256, &bad).unwrap();
        plan.update_gather(&mut data).unwrap();
        assert!(matches!(
            plan.update_factor(&mut data),
            Err(TransferError::InvalidPivot {
                world: 256,
                dof: 65,
                value
            }) if value == -1.0
        ));
        assert!(data.readback().is_err());
        data.write_world_matrix(256, &m1).unwrap();
        plan.update(&mut data).unwrap();
        let snapshot = data.readback().unwrap();
        assert!(
            engine::physics::SparseMassFactorPlan::new(&session, input::mixed_input()).is_err()
        );
        assert!(
            SparseMassSmallPlan::new(&session, input::sparse(&input::base()["models"][3])).is_err()
        );
        drop(data);
        drop(plan);
        drop(foreign);
        drop(session);
        assert_eq!(snapshot.worlds(), 257);
        assert!(snapshot.world(257).is_err());
        let session = TransferSession::new(0).unwrap();
        use engine::model::{SparseMassFields, SparseMassInput};
        let compact = SparseMassInput::new(SparseMassFields {
            dof_parentid: vec![-1],
            m_rownnz: vec![1],
            m_rowadr: vec![0],
            m_colind: vec![0],
            tree_dofadr: vec![0],
            tree_dofnum: vec![1],
        })
        .unwrap();
        for model in [
            SparseMassInput::new(SparseMassFields::default()).unwrap(),
            compact,
            input::sparse(&input::base()["models"][0]),
        ] {
            let nc = model.nc();
            let plan = SparseMassSmallPlan::new(&session, model).unwrap();
            let mut data = plan.create_data(2).unwrap();
            data.write_matrix(&vec![1.0; 2 * nc]).unwrap();
            plan.update(&mut data).unwrap();
            assert!(data.readback().unwrap().world(1).unwrap().is_empty());
        }
    }
